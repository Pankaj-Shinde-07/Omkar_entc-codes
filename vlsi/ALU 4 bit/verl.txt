module tb_t_flipflop;

reg T;
reg clk;
reg reset;
wire Q;

t_flipflop uut (
    .T(T),
    .clk(clk),
    .reset(reset),
    .Q(Q)
);

initial begin
    clk = 0;
    forever #5 clk = ~clk;
end

initial begin
    reset = 1;
    T = 0;
    #10 reset = 0;

    T = 1;
    #20;

    T = 0;
    #20;

    T = 1;
    #20;

    $finish;
end

initial begin
    $monitor("Time=%0t T=%b Q=%b", $time, T, Q);
end

endmodule

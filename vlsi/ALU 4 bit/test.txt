
module alu4bit_tb;

reg [3:0] A, B;
reg [2:0] sel;
wire [3:0] Y;
wire Cout;

// Instantiate ALU
alu4bit uut(
    .A(A),
    .B(B),
    .sel(sel),
    .Y(Y),
    .Cout(Cout)
);

initial begin

    $monitor("A=%b B=%b sel=%b | Y=%b Cout=%b",
              A, B, sel, Y, Cout);

    A = 4'b0101;
    B = 4'b0011;

    sel = 3'b000; #10;  // Addition
    sel = 3'b001; #10;  // Subtraction
    sel = 3'b010; #10;  // AND
    sel = 3'b011; #10;  // OR
    sel = 3'b100; #10;  // XOR
    sel = 3'b101; #10;  // NOT
    sel = 3'b110; #10;  // Left Shift
    sel = 3'b111; #10;  // Right Shift

    $finish;

end

endmodule

module full_adder_tb;

reg A,B,Cin;
wire sum,cout;

full_adder DUT(
	.A(A),
	.B(B),
	.cin(Cin),
	.sum(sum),
	.cout(cout)
);

initial begin
	$monitor("Time=%0t | A=%b B=%b Cin=%b | sum=%b cout=%b",
					$time,A,B,Cin,sum,cout);
	A=0;B=0; Cin=0;
	#10;
	A=0;B=0; Cin=1;
	#10;
	A=0;B=1; Cin=0;
	#10;
	A=0;B=1; Cin=1;
	#10;
	A=1;B=0; Cin=0;
	#10;
	A=1;B=0; Cin=1;
	#10;
	A=1;B=1; Cin=0;
	#10;
	A=1;B=1; Cin=1;
	#10;
	
	$finish;
	
	end 
	endmodule

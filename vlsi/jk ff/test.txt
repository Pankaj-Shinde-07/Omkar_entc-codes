`timescale 1ns/1ps

module jk_flip_flop_tb;

reg clk,rst,J,K;
wire Q;

jk_flip_flop uut(
	.clk(clk),
	.rst(rst),
	.J(J),
	.K(K),
	.Q(Q)
);
always #5 clk=~clk;

initial begin
	$monitor("Time=%0t rst=%b J=%b K=%b Q=%b",
					$time,rst,J,K,Q);
	clk=0;
	rst=1;
	J=0;
	K=0;
	
	#10 rst=0;
	
	#10 J=0;K=0;
	#10 J=0;K=1;
	#10 J=1;K=0;
	#10 J=1;K=1;
	
	#10 $finish;
end

endmodule
